Checkbox

A control that allows the user to toggle between checked and not checked.

Default

<div class="flex items-center gap-3">
  <button
    type="button"
    role="checkbox"
    aria-checked="false"
    data-state="unchecked"
    id="checkbox-default"
    class="peer size-4 shrink-0 rounded-[4px] border border-input dark:bg-input/30 shadow-xs transition-all outline-none focus-visible:ring-[3px] focus-visible:ring-ring/50 focus-visible:border-ring disabled:cursor-not-allowed disabled:opacity-50 data-[state=checked]:bg-primary data-[state=checked]:text-primary-foreground data-[state=checked]:border-primary dark:data-[state=checked]:bg-primary"
    onclick="toggleCheckbox(this)"
  >
    <svg
      class="hidden size-4 text-current"
      xmlns="http://www.w3.org/2000/svg"
      viewBox="0 0 24 24"
      fill="none"
      stroke="currentColor"
      stroke-width="3"
      stroke-linecap="round"
      stroke-linejoin="round"
    >
      <polyline points="20 6 9 17 4 12"/>
    </svg>
  </button>
  <label
    for="checkbox-default"
    class="text-sm leading-none font-medium select-none peer-disabled:cursor-not-allowed peer-disabled:opacity-50"
  >
    Accept terms and conditions
  </label>
</div>

<script>
function toggleCheckbox(btn) {
  const isChecked = btn.getAttribute('data-state') === 'checked';
  const newState = isChecked ? 'unchecked' : 'checked';
  btn.setAttribute('data-state', newState);
  btn.setAttribute('aria-checked', !isChecked);
  
  const svg = btn.querySelector('svg');
  if (newState === 'checked') {
    svg.classList.remove('hidden');
  } else {
    svg.classList.add('hidden');
  }
}
</script>
