Tooltip

A popup that displays information related to an element when the element receives keyboard focus or the mouse hovers over it.

Default (Top)

Add to library 
<div class="tooltip-wrapper">
  <button
    onmouseenter="showTooltip('tooltip-1', 'top')"
    onmouseleave="hideTooltip('tooltip-1')"
  >
    Hover me
  </button>
  <div
    id="tooltip-1"
    class="tooltip-content absolute z-50 overflow-hidden rounded-md bg-primary px-3 py-1.5 text-xs text-primary-foreground hidden"
    data-state="closed"
    role="tooltip"
  >
    Add to library
  </div>
</div>

Delayed Show

Tooltip appears after a delay (default: 700ms)

I appear after a delay 
// Show with 700ms delay
showTooltip('id', 'top', 700);

JavaScript

let tooltipTimeouts = {};

function showTooltip(id, position = 'top', delay = 0) {
  // Clear existing timeout
  if (tooltipTimeouts[id]) {
    clearTimeout(tooltipTimeouts[id]);
  }

  // Set new timeout
  tooltipTimeouts[id] = setTimeout(() => {
    const tooltip = document.getElementById(id);
    const wrapper = tooltip.parentElement;
    const trigger = wrapper.querySelector('button');
    
    // Remove all position classes
    tooltip.classList.remove('top-full', 'bottom-full', 'left-full', 'right-full');
    tooltip.classList.remove('left-1/2', '-translate-x-1/2');
    tooltip.classList.remove('top-1/2', '-translate-y-1/2');
    tooltip.classList.remove('mt-2', 'mb-2', 'ml-2', 'mr-2');
    
    // Add position classes based on direction BEFORE showing
    if (position === 'top') {
      tooltip.classList.add('bottom-full', 'left-1/2', '-translate-x-1/2', 'mb-2');
    } else if (position === 'bottom') {
      tooltip.classList.add('top-full', 'left-1/2', '-translate-x-1/2', 'mt-2');
    } else if (position === 'left') {
      tooltip.classList.add('right-full', 'top-1/2', '-translate-y-1/2', 'mr-2');
    } else if (position === 'right') {
      tooltip.classList.add('left-full', 'top-1/2', '-translate-y-1/2', 'ml-2');
    }
    
    // Show tooltip AFTER position is set
    tooltip.classList.remove('hidden');
    tooltip.classList.add('block');
    
    // Trigger animation slightly after display change
    requestAnimationFrame(() => {
      tooltip.setAttribute('data-state', 'open');
    });
  }, delay);
}

function hideTooltip(id) {
  // Clear timeout
  if (tooltipTimeouts[id]) {
    clearTimeout(tooltipTimeouts[id]);
    delete tooltipTimeouts[id];
  }
  
  const tooltip = document.getElementById(id);
  if (!tooltip) return;
  
  tooltip.setAttribute('data-state', 'closed');
  
  // Hide after animation
  setTimeout(() => {
    if (tooltip.getAttribute('data-state') === 'closed') {
      tooltip.classList.add('hidden');
      tooltip.classList.remove('block');
    }
  }, 100);
}

CSS Animations

/* Tooltip animations */
@keyframes tooltip-fade-in {
  from {
    opacity: 0;
    transform: scale(0.96);
  }
  to {
    opacity: 1;
    transform: scale(1);
  }
}

@keyframes tooltip-fade-out {
  from {
    opacity: 1;
    transform: scale(1);
  }
  to {
    opacity: 0;
    transform: scale(0.96);
  }
}

.tooltip-content[data-state="open"] {
  animation: tooltip-fade-in 0.15s ease-out;
}

.tooltip-content[data-state="closed"] {
  animation: tooltip-fade-out 0.1s ease-in;
}
