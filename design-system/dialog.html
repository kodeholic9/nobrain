Dialog

A window overlaid on either the primary window or another dialog window, rendering the content underneath inert.

Default

Are you absolutely sure?

This action cannot be undone. 

<!-- Trigger Button -->
<button onclick="openDialog('dialog-1')">Open Dialog</button>

<!-- Dialog -->
<div id="dialog-1" class="dialog-container" data-state="closed">
  <!-- Overlay -->
  <div
    class="dialog-overlay fixed inset-0 z-50 bg-black/80"
    onclick="closeDialog('dialog-1')"
  ></div>

  <!-- Content -->
  <div class="dialog-content fixed left-[50%] top-[50%] z-50 grid w-full max-w-lg translate-x-[-50%] translate-y-[-50%] gap-4 border bg-background p-6 shadow-lg sm:rounded-lg">
    <!-- Close Button (X) -->
    <button onclick="closeDialog('dialog-1')" class="absolute right-4 top-4 ...">
      <svg ...>X Icon</svg>
    </button>

    <!-- Header -->
    <div>
      <h2>Are you absolutely sure?</h2>
      <p>This action cannot be undone.</p>
    </div>

    <!-- Footer -->
    <div class="flex flex-col-reverse sm:flex-row sm:justify-end sm:space-x-2">
      <button>Cancel</button>
      <button>Continue</button>
    </div>
  </div>
</div>

JavaScript

function openDialog(id) {
  const dialog = document.getElementById(id);
  const overlay = dialog.querySelector('.dialog-overlay');
  const content = dialog.querySelector('.dialog-content');
  
  dialog.style.display = 'block';
  dialog.setAttribute('data-state', 'open');
  
  // Trigger reflow for animation
  void overlay.offsetWidth;
  void content.offsetWidth;
  
  overlay.setAttribute('data-state', 'open');
  content.setAttribute('data-state', 'open');
  
  // Prevent body scroll
  document.body.style.overflow = 'hidden';
}

function closeDialog(id) {
  const dialog = document.getElementById(id);
  const overlay = dialog.querySelector('.dialog-overlay');
  const content = dialog.querySelector('.dialog-content');
  
  overlay.setAttribute('data-state', 'closed');
  content.setAttribute('data-state', 'closed');
  dialog.setAttribute('data-state', 'closed');
  
  // Wait for animation to finish
  setTimeout(() => {
    dialog.style.display = 'none';
    document.body.style.overflow = '';
  }, 150);
}

// Close on Escape key
document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape') {
    const openDialogs = document.querySelectorAll('.dialog-container[data-state="open"]');
    openDialogs.forEach(dialog => {
      closeDialog(dialog.id);
    });
  }
});
