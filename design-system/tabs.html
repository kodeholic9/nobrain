Tabs

A set of layered sections of content—known as tab panels—that are displayed one at a time.

Default

Make changes to your account here. Click save when you're done.

Change your password here. After saving, you'll be logged out. 

<div class="w-full" data-tabs="default">
  <!-- Tabs List -->
  <div class="inline-flex h-9 items-center justify-center rounded-lg bg-muted p-1 text-muted-foreground">
    <button
      data-tab-trigger="account"
      class="inline-flex items-center justify-center whitespace-nowrap rounded-md px-3 py-1 text-sm font-medium transition-all data-[state=active]:bg-background data-[state=active]:text-foreground data-[state=active]:shadow"
      data-state="active"
      onclick="switchTab('default', 'account')"
    >
      Account
    </button>
    <button
      data-tab-trigger="password"
      class="inline-flex items-center justify-center whitespace-nowrap rounded-md px-3 py-1 text-sm font-medium transition-all data-[state=active]:bg-background data-[state=active]:text-foreground data-[state=active]:shadow"
      data-state="inactive"
      onclick="switchTab('default', 'password')"
    >
      Password
    </button>
  </div>

  <!-- Tabs Content -->
  <div class="mt-4">
    <div data-tab-content="account" class="block">
      <p>Make changes to your account here.</p>
    </div>
    <div data-tab-content="password" class="hidden">
      <p>Change your password here.</p>
    </div>
  </div>
</div>

<script>
function switchTab(groupId, tabName) {
  const group = document.querySelector(`[data-tabs="${groupId}"]`);
  
  // Update triggers
  group.querySelectorAll('[data-tab-trigger]').forEach(trigger => {
    trigger.setAttribute('data-state', 'inactive');
  });
  group.querySelector(`[data-tab-trigger="${tabName}"]`).setAttribute('data-state', 'active');
  
  // Update content
  group.querySelectorAll('[data-tab-content]').forEach(content => {
    content.classList.add('hidden');
    content.classList.remove('block');
  });
  group.querySelector(`[data-tab-content="${tabName}"]`).classList.remove('hidden');
  group.querySelector(`[data-tab-content="${tabName}"]`).classList.add('block');
}
</script>
