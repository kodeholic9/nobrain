Custom Select (Tailwind Only)

Fully customizable select component using only Tailwind utility classes.

Basic

Apple
Banana
Orange
Grape
Mango 
<div id="select-fruit" class="relative w-full">
  <button type="button" class="flex items-center justify-between w-full h-9 px-3 py-1 border border-input rounded-md bg-transparent text-sm transition-colors hover:bg-accent focus:outline-none focus:ring-1 focus:ring-ring cursor-pointer" aria-expanded="false">
    <span class="text-muted-foreground">Select a fruit</span>
    <svg class="w-4 h-4 text-muted-foreground transition-transform" xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
      <polyline points="6 9 12 15 18 9"/>
    </svg>
  </button>
  <div class="hidden absolute top-full mt-1 left-0 right-0 z-50 max-h-60 overflow-y-auto border border-border rounded-md bg-popover shadow-md">
    <div class="px-3 py-2 text-sm text-popover-foreground cursor-pointer hover:bg-accent hover:text-accent-foreground transition-colors" data-value="apple">Apple</div>
    <div class="px-3 py-2 text-sm text-popover-foreground cursor-pointer hover:bg-accent hover:text-accent-foreground transition-colors" data-value="banana">Banana</div>
    <div class="px-3 py-2 text-sm text-popover-foreground cursor-pointer hover:bg-accent hover:text-accent-foreground transition-colors" data-value="orange">Orange</div>
    <div class="px-3 py-2 text-sm text-popover-foreground cursor-pointer hover:bg-accent hover:text-accent-foreground transition-colors" data-value="grape">Grape</div>
    <div class="px-3 py-2 text-sm text-popover-foreground cursor-pointer hover:bg-accent hover:text-accent-foreground transition-colors" data-value="mango">Mango</div>
  </div>
</div>

<script>
function initCustomSelect(selectId) {
  const wrapper = document.getElementById(selectId);
  if (!wrapper) return;
  
  const trigger = wrapper.querySelector('button[aria-expanded]');
  const dropdown = wrapper.querySelector('.hidden.absolute');
  const valueEl = trigger.querySelector('span');
  const icon = trigger.querySelector('svg');
  const options = wrapper.querySelectorAll('[data-value]');

  trigger.addEventListener('click', () => {
    const isOpen = trigger.getAttribute('aria-expanded') === 'true';
    trigger.setAttribute('aria-expanded', !isOpen);
    dropdown.classList.toggle('hidden');
    icon.style.transform = isOpen ? 'rotate(0deg)' : 'rotate(180deg)';
  });

  options.forEach(option => {
    option.addEventListener('click', () => {
      const text = option.textContent;
      
      options.forEach(opt => opt.classList.remove('bg-accent', 'font-medium'));
      option.classList.add('bg-accent', 'font-medium');
      
      valueEl.textContent = text;
      valueEl.classList.remove('text-muted-foreground');
      valueEl.classList.add('text-foreground');
      
      trigger.setAttribute('aria-expanded', 'false');
      dropdown.classList.add('hidden');
      icon.style.transform = 'rotate(0deg)';
    });
  });

  document.addEventListener('click', (e) => {
    if (!wrapper.contains(e.target)) {
      trigger.setAttribute('aria-expanded', 'false');
      dropdown.classList.add('hidden');
      icon.style.transform = 'rotate(0deg)';
    }
  });
}

// Initialize
initCustomSelect('select-fruit');
</script>
