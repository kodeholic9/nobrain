<script>
document.getElementById('file-preview-input').addEventListener('change', function(e) {
  const file = e.target.files[0];
  if (file && file.type.startsWith('image/')) {
    const reader = new FileReader();
    reader.onload = function(e) {
      document.getElementById('preview-image').src = e.target.result;
      document.getElementById('preview-container').classList.remove('hidden');
    };
    reader.readAsDataURL(file);
  }
});
</script>

  />
</div>
<div id="dropzone-preview" class="mt-4 hidden">
  <div class="text-sm font-medium mb-2">Selected file:</div>
  <div class="flex items-center gap-4 p-4 border rounded-lg">
    <img id="dropzone-thumbnail" class="h-20 w-20 object-cover rounded" />
    <div class="flex-1">
      <p id="dropzone-filename" class="text-sm font-medium"></p>
      <p id="dropzone-filesize" class="text-xs text-muted-foreground"></p>
    </div>
    <button
      id="dropzone-remove"
      class="text-destructive hover:text-destructive/80 text-sm"
    >
      Remove
    </button>
  </div>
</div>

<script>
const dropzone = document.getElementById('dropzone');
const dropzoneInput = document.getElementById('dropzone-input');
const dropzonePreview = document.getElementById('dropzone-preview');
const dropzoneThumbnail = document.getElementById('dropzone-thumbnail');
const dropzoneFilename = document.getElementById('dropzone-filename');
const dropzoneFilesize = document.getElementById('dropzone-filesize');
const dropzoneRemove = document.getElementById('dropzone-remove');

// Click to upload
dropzone.addEventListener('click', () => dropzoneInput.click());

// Handle file selection
dropzoneInput.addEventListener('change', handleFiles);

// Drag and drop handlers
dropzone.addEventListener('dragover', (e) => {
  e.preventDefault();
  dropzone.classList.add('border-primary');
});

dropzone.addEventListener('dragleave', () => {
  dropzone.classList.remove('border-primary');
});

dropzone.addEventListener('drop', (e) => {
  e.preventDefault();
  dropzone.classList.remove('border-primary');
  const files = e.dataTransfer.files;
  if (files.length) {
    dropzoneInput.files = files;
    handleFiles();
  }
});

function handleFiles() {
  const file = dropzoneInput.files[0];
  if (file && file.type.startsWith('image/')) {
    const reader = new FileReader();
    reader.onload = (e) => {
      dropzoneThumbnail.src = e.target.result;
      dropzoneFilename.textContent = file.name;
      dropzoneFilesize.textContent = (file.size / 1024).toFixed(2) + ' KB';
      dropzonePreview.classList.remove('hidden');
    };
    reader.readAsDataURL(file);
  }
}

// Remove file
dropzoneRemove.addEventListener('click', (e) => {
  e.stopPropagation();
  dropzoneInput.value = '';
  dropzonePreview.classList.add('hidden');
});
</script>
