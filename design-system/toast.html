Toast

An opinionated toast component using Toastify.js - a lightweight vanilla JavaScript library.

Installation

<!-- Add Toastify CSS -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/npm/toastify-js/src/toastify.min.css">

<!-- Add Toastify JS -->
<script src="https://cdn.jsdelivr.net/npm/toastify-js"></script>

Simple

Toastify({
  text: "Event has been created",
  duration: 4000,
  gravity: "bottom",
  position: "right"
}).showToast();

Different Types (Responsive) 

// Success
Toastify({
  text: "Event has been created",
  className: "toast-success",
  duration: 4000
}).showToast();

// Error
Toastify({
  text: "Something went wrong",
  className: "toast-error",
  duration: 4000
}).showToast();

// Warning
Toastify({
  text: "Please check your input",
  className: "toast-warning",
  duration: 4000
}).showToast();

// Info
Toastify({
  text: "New message received",
  className: "toast-info",
  duration: 4000
}).showToast();

Positions (Responsive) 

// Top Left
Toastify({
  text: "Notification",
  gravity: "top",
  position: "left"
}).showToast();

// Bottom Center
Toastify({
  text: "Notification",
  gravity: "bottom",
  position: "center"
}).showToast();

With Callback

Toastify({
  text: "Click me to trigger action",
  duration: 4000,
  onClick: function() {
    console.log('Toast clicked!');
    // Perform action here
  }
}).showToast();

Custom Duration

// 2 seconds
Toastify({
  text: "Short notification",
  duration: 2000
}).showToast();

// 5 seconds
Toastify({
  text: "Longer notification",
  duration: 5000
}).showToast();

// No auto-close (requires manual close)
Toastify({
  text: "Manual close required",
  duration: -1,
  close: true // Show close button
}).showToast();

Advanced Options

Toastify({
  text: "Notification message",
  duration: 4000,
  
  // Position
  gravity: "top",        // "top" or "bottom"
  position: "right",     // "left", "center", "right"
  
  // Styling
  className: "toast-success",  // Custom class
  style: {
    background: "linear-gradient(to right, #00b09b, #96c93d)",
  },
  
  // Offset
  offset: {
    x: 50,  // horizontal offset
    y: 10   // vertical offset
  },
  
  // Close button
  close: true,           // Show close button
  
  // Callbacks
  onClick: function() {  // Click handler
    console.log('Clicked!');
  },
  callback: function() { // After dismiss
    console.log('Dismissed!');
  },
  
  // Behavior
  stopOnFocus: true,     // Pause auto-dismiss on hover
  escapeMarkup: true,    // Escape HTML
  oldestFirst: true      // Stack order
}).showToast();

Helper Functions (Optional) 

// Create helper functions for common toast types
function toast(text, options = {}) {
  return Toastify({
    text: text,
    duration: 4000,
    gravity: "bottom",
    position: "right",
    ...options
  }).showToast();
}

function toastSuccess(text) {
  return toast(text, { className: "toast-success" });
}

function toastError(text) {
  return toast(text, { className: "toast-error" });
}

function toastWarning(text) {
  return toast(text, { className: "toast-warning" });
}

function toastInfo(text) {
  return toast(text, { className: "toast-info" });
}

// Usage
toastSuccess("Saved successfully!");
toastError("Failed to save");
toastWarning("Please check your input");
toastInfo("New update available");
