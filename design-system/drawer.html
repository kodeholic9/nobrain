Drawer

A drawer component for mobile devices that slides up from the bottom.

Default

Are you absolutely sure?

This action cannot be undone.

        </p>

  </div>
</div>

JavaScript

function openDrawer(id) {
  const drawer = document.getElementById(id);
  const overlay = drawer.querySelector('.drawer-overlay');
  const content = drawer.querySelector('.drawer-content');
  
  drawer.style.display = 'block';
  drawer.setAttribute('data-state', 'open');
  
  // Trigger reflow for animation
  void overlay.offsetWidth;
  void content.offsetWidth;
  
  overlay.setAttribute('data-state', 'open');
  content.setAttribute('data-state', 'open');
  
  // Prevent body scroll
  document.body.style.overflow = 'hidden';
}

function closeDrawer(id) {
  const drawer = document.getElementById(id);
  const overlay = drawer.querySelector('.drawer-overlay');
  const content = drawer.querySelector('.drawer-content');
  
  overlay.setAttribute('data-state', 'closed');
  content.setAttribute('data-state', 'closed');
  drawer.setAttribute('data-state', 'closed');
  
  // Wait for animation to finish
  setTimeout(() => {
    drawer.style.display = 'none';
    document.body.style.overflow = '';
  }, 300);
}

// Close on Escape key
document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape') {
    const openDrawers = document.querySelectorAll('.drawer-container[data-state="open"]');
    openDrawers.forEach(drawer => {
      closeDrawer(drawer.id);
    });
  }
});

CSS Animations

@keyframes drawer-slide-up {
  from { transform: translateY(100%); }
  to { transform: translateY(0); }
}

@keyframes drawer-slide-down {
  from { transform: translateY(0); }
  to { transform: translateY(100%); }
}

@keyframes overlay-fade-in {
  from { opacity: 0; }
  to { opacity: 1; }
}

@keyframes overlay-fade-out {
  from { opacity: 1; }
  to { opacity: 0; }
}

.drawer-overlay[data-state="open"] {
  animation: overlay-fade-in 0.2s ease-out;
}

.drawer-overlay[data-state="closed"] {
  animation: overlay-fade-out 0.2s ease-out;
}

.drawer-content[data-state="open"] {
  animation: drawer-slide-up 0.3s cubic-bezier(0.32, 0.72, 0, 1);
}

.drawer-content[data-state="closed"] {
  animation: drawer-slide-down 0.3s cubic-bezier(0.32, 0.72, 0, 1);
}
