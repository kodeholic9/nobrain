With Character Count

0 / 200 

<div class="space-y-2">
  <label class="text-sm font-medium leading-none">
    Comment
  </label>
  <textarea
    id="textarea-counter"
    maxlength="200"
    placeholder="Write a comment..."
    class="flex min-h-[80px] w-full rounded-md border border-input bg-transparent px-3 py-2 text-sm shadow-xs placeholder:text-muted-foreground focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring disabled:cursor-not-allowed disabled:opacity-50"
  ></textarea>
  <p class="text-sm text-muted-foreground text-right">
    <span id="char-count">0</span> / 200
  </p>
</div>

<script>
const textarea = document.getElementById('textarea-counter');
const charCount = document.getElementById('char-count');

textarea.addEventListener('input', () => {
  charCount.textContent = textarea.value.length;
});
</script>

Auto-growing

Automatically adjusts height based on content. 

<textarea
  id="auto-grow-textarea"
  placeholder="Start typing... (grows automatically)"
  class="flex min-h-[80px] w-full rounded-md border border-input bg-transparent px-3 py-2 text-sm shadow-xs placeholder:text-muted-foreground focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring disabled:cursor-not-allowed disabled:opacity-50 resize-none overflow-hidden"
  rows="3"
></textarea>

<script>
const autoGrowTextarea = document.getElementById('auto-grow-textarea');

autoGrowTextarea.addEventListener('input', function() {
  this.style.height = 'auto';
  this.style.height = this.scrollHeight + 'px';
});
</script>
