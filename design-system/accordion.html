JavaScript

function toggleAccordion(accordionId, itemIndex) {
  const accordion = document.querySelector(`[data-accordion="${accordionId}"]`);
  const type = accordion.getAttribute('data-type') || 'multiple';
  const items = accordion.querySelectorAll('.accordion-item');
  const currentItem = items[itemIndex];
  
  const trigger = currentItem.querySelector('.accordion-trigger');
  const content = currentItem.querySelector('.accordion-content');
  const currentState = currentItem.getAttribute('data-state');
  
  // Type: single - close other items
  if (type === 'single') {
    items.forEach((item, index) => {
      if (index !== itemIndex) {
        const otherTrigger = item.querySelector('.accordion-trigger');
        const otherContent = item.querySelector('.accordion-content');
        
        item.setAttribute('data-state', 'closed');
        otherTrigger.setAttribute('data-state', 'closed');
        otherContent.setAttribute('data-state', 'closed');
        otherContent.classList.add('hidden');
        otherContent.classList.remove('block');
      }
    });
  }
  
  // Toggle current item
  if (currentState === 'closed') {
    currentItem.setAttribute('data-state', 'open');
    trigger.setAttribute('data-state', 'open');
    content.setAttribute('data-state', 'open');
    content.classList.remove('hidden');
    content.classList.add('block');
  } else {
    currentItem.setAttribute('data-state', 'closed');
    trigger.setAttribute('data-state', 'closed');
    content.setAttribute('data-state', 'closed');
    
    // Delay hiding to allow animation
    setTimeout(() => {
      if (content.getAttribute('data-state') === 'closed') {
        content.classList.add('hidden');
        content.classList.remove('block');
      }
    }, 200);
  }
}

CSS Animations

/* Accordion animations */
@keyframes accordion-down {
  from {
    height: 0;
    opacity: 0;
  }
  to {
    height: var(--radix-accordion-content-height);
    opacity: 1;
  }
}

@keyframes accordion-up {
  from {
    height: var(--radix-accordion-content-height);
    opacity: 1;
  }
  to {
    height: 0;
    opacity: 0;
  }
}

.accordion-content[data-state="open"] {
  animation: accordion-down 0.2s ease-out;
}

.accordion-content[data-state="closed"] {
  animation: accordion-up 0.2s ease-out;
}

/* Chevron rotation */
.accordion-trigger svg {
  transition: transform 0.2s ease-out;
}

.accordion-trigger[data-state="open"] svg {
  transform: rotate(180deg);
}
